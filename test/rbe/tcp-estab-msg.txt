
func TestEstablishedICMPPoll4(t *testing.T) {
	for _, enabled := range []int32{0, 1} {
		for _, msgFirst := range []bool{false, true} {
			t.Run(fmt.Sprintf("enabled_%d_msgfirst_%t", enabled, msgFirst), func(t *testing.T) {
				dut := testbench.NewDUT(t)
				listener, port := dut.CreateListener(t, unix.SOCK_STREAM, unix.IPPROTO_TCP, 1)
				defer dut.Close(t, listener)
				conn := dut.Net.NewTCPIPv4(t, testbench.TCP{DstPort: &port}, testbench.TCP{SrcPort: &port})
				defer conn.Close(t)
				conn.Connect(t)
				fd, _ := dut.Accept(t, listener)
				defer dut.Close(t, fd)
				dut.SetSockOptInt(t, fd, unix.SOL_IP, unix.IP_RECVERR, enabled)
				pollerr := func(wait time.Duration) int16 {
					fds := dut.Poll(t, []unix.PollFd{{Fd: fd, Events: unix.POLLERR}}, wait)
					if len(fds) == 0 {
						return 0
					}
					return fds[0].Revents
				}
				if got := pollerr(0); got != 0 {
					t.Fatalf("dirty fresh socket: %d", got)
				}
				dut.Send(t, fd, []byte("quote me"), 0)
				frame, err := conn.ExpectData(t, &testbench.TCP{}, &testbench.Payload{Bytes: []byte("quote me")}, time.Second)
				if err != nil {
					t.Fatalf("TCP payload to quote: %v", err)
				}
				sendICMPError(t, &conn, frame[len(frame)-2].(*testbench.TCP))
				signals := pollerr(time.Second)
				var soError int32
				var afterFirst int16
				if !msgFirst {
					soError = dut.GetSockOptInt(t, fd, unix.SOL_SOCKET, unix.SO_ERROR)
					afterFirst = pollerr(0)
				}
				msg := dut.RecvErrMessage(t, fd)
				var level, typ int32
				var ee uint32
				var origin, icmpType, icmpCode byte
				if msg.GetRet() >= 0 {
					if int(msg.GetRet()) != len(msg.GetBuf()) || msg.GetMsgFlags()&(unix.MSG_TRUNC|unix.MSG_CTRUNC) != 0 {
						t.Fatalf("truncated recvmsg result: %d/%d, flags %#x", msg.GetRet(), len(msg.GetBuf()), msg.GetMsgFlags())
					}
					cms, err := unix.ParseSocketControlMessage(msg.GetControl())
					if err != nil || len(cms) != 1 {
						t.Fatalf("recvmsg ancillary data: %d bytes, messages=%v, err=%v", len(msg.GetControl()), cms, err)
					}
					c := cms[0]
					level, typ = c.Header.Level, c.Header.Type
					if level != unix.IPPROTO_IP || typ != unix.IP_RECVERR || len(c.Data) < 16 {
						t.Fatalf("unexpected control level=%d type=%d len=%d", level, typ, len(c.Data))
					}
					ee = binary.NativeEndian.Uint32(c.Data[:4])
					origin, icmpType, icmpCode = c.Data[4], c.Data[5], c.Data[6]
					if ee != uint32(unix.EHOSTUNREACH) || origin != 2 || icmpType != 3 || icmpCode != 1 || msg.GetMsgFlags()&unix.MSG_ERRQUEUE == 0 {
						t.Fatalf("wrong extended ICMP error: flags %#x errno/origin/type/code=%d/%d/%d/%d", msg.GetMsgFlags(), ee, origin, icmpType, icmpCode)
					}
				}
				if msgFirst {
					afterFirst = pollerr(0)
					soError = dut.GetSockOptInt(t, fd, unix.SOL_SOCKET, unix.SO_ERROR)
				}
				empty := dut.RecvErrMessage(t, fd)
				if empty.GetRet() != -1 || empty.GetErrno_() != int32(unix.EAGAIN) {
					t.Fatalf("repeat recvmsg: n=%d errno=%d", empty.GetRet(), empty.GetErrno_())
				}
				t.Logf("ESTAB_MSG family=4 opt=%d first=%t poll=%d so=%d afterfirst=%d n=%d errno=%d flags=%d cbytes=%d level=%d type=%d ee=%d origin=%d itype=%d code=%d again=%d final=%d", enabled, msgFirst, signals, soError, afterFirst, msg.GetRet(), msg.GetErrno_(), msg.GetMsgFlags(), len(msg.GetControl()), level, typ, ee, origin, icmpType, icmpCode, empty.GetErrno_(), pollerr(0))
				conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagAck)})
				dut.Send(t, fd, []byte("still connected"), 0)
				if _, err := conn.ExpectData(t, &testbench.TCP{}, &testbench.Payload{Bytes: []byte("still connected")}, 2*time.Second); err != nil {
					t.Fatalf("connection did not carry fresh application data: %v", err)
				}
				t.Logf("ESTAB_MSG_LIVE family=4 opt=%d first=%t", enabled, msgFirst)
			})
		}
	}
}

func TestEstablishedICMPData4(t *testing.T) {
	for _, op := range []string{"read", "write"} {
		t.Run(op, func(t *testing.T) {
			dut := testbench.NewDUT(t)
			listener, port := dut.CreateListener(t, unix.SOCK_STREAM, unix.IPPROTO_TCP, 1)
			defer dut.Close(t, listener)
			conn := dut.Net.NewTCPIPv4(t, testbench.TCP{DstPort: &port}, testbench.TCP{SrcPort: &port})
			defer conn.Close(t)
			conn.Connect(t)
			fd, _ := dut.Accept(t, listener)
			defer dut.Close(t, fd)
			dut.SetSockOptInt(t, fd, unix.SOL_IP, unix.IP_RECVERR, 1)
			pollerr := func(wait time.Duration) int16 {
				fds := dut.Poll(t, []unix.PollFd{{Fd: fd, Events: unix.POLLERR}}, wait)
				if len(fds) == 0 {
					return 0
				}
				return fds[0].Revents
			}
			if got := pollerr(0); got != 0 {
				t.Fatalf("dirty fresh socket: %d", got)
			}
			dut.Send(t, fd, []byte("quote me"), 0)
			frame, err := conn.ExpectData(t, &testbench.TCP{}, &testbench.Payload{Bytes: []byte("quote me")}, time.Second)
			if err != nil {
				t.Fatalf("TCP payload to quote: %v", err)
			}
			sendICMPError(t, &conn, frame[len(frame)-2].(*testbench.TCP))

			poll := pollerr(time.Second)
			var n int32
			var ioerr error
			if op == "read" {
				n, _, ioerr = dut.RecvWithErrno(context.Background(), t, fd, 1, unix.MSG_DONTWAIT)
			} else {
				n, ioerr = dut.SendWithErrno(context.Background(), t, fd, []byte("x"), unix.MSG_DONTWAIT)
			}
			so := dut.GetSockOptInt(t, fd, unix.SOL_SOCKET, unix.SO_ERROR)
			after := pollerr(0)
			got := 0
			if ioerr != nil {
				got = int(ioerr.(unix.Errno))
			}
			t.Logf("ICMP_IO family=4 op=%s before=%d n=%d errno=%d after=%d so=%d", op, poll, n, got, after, so)
			if poll != unix.POLLERR || n != -1 || got != 113 || after != 0 || so != 0 {
				t.Fatalf("pending error not returned exactly once")
			}
			conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagAck)})
			dut.Send(t, fd, []byte("still connected"), 0)
			if _, err := conn.ExpectData(t, &testbench.TCP{}, &testbench.Payload{Bytes: []byte("still connected")}, 2*time.Second); err != nil {
				t.Fatalf("connection did not carry fresh application data: %v", err)
			}

			t.Logf("ICMP_IO_LIVE family=4 op=%s", op)
		})
	}
}

func TestEstablishedICMPStaleQuote(t *testing.T) {
	dut := testbench.NewDUT(t)
	listener, port := dut.CreateListener(t, unix.SOCK_STREAM, unix.IPPROTO_TCP, 1)
	defer dut.Close(t, listener)
	conn := dut.Net.NewTCPIPv4(t, testbench.TCP{DstPort: &port}, testbench.TCP{SrcPort: &port})
	defer conn.Close(t)
	conn.Connect(t)
	fd, _ := dut.Accept(t, listener)
	defer dut.Close(t, fd)
	dut.SetSockOptInt(t, fd, unix.SOL_IP, unix.IP_RECVERR, 1)
	send := func(s string) *testbench.TCP {
		t.Helper()
		dut.Send(t, fd, []byte(s), 0)
		frame, err := conn.ExpectData(t, &testbench.TCP{}, &testbench.Payload{Bytes: []byte(s)}, 2*time.Second)
		if err != nil { t.Fatalf("capture %q: %v", s, err) }
		return frame[len(frame)-2].(*testbench.TCP)
	}
	barrier := func(marker string) {
		t.Helper()
		conn.Send(t, testbench.TCP{Flags:testbench.TCPFlags(header.TCPFlagAck | header.TCPFlagPsh)}, &testbench.Payload{Bytes:[]byte(marker)})
		if got := string(dut.Recv(t, fd, int32(len(marker)), unix.MSG_WAITALL)); got != marker { t.Fatalf("ACK-bearing packet delivered %q, want %q", got, marker) }
	}
	poll := func() int16 {
		fds:=dut.Poll(t, []unix.PollFd{{Fd: fd, Events: unix.POLLERR}}, 250*time.Millisecond)
		if len(fds)==0 { return 0 }
		return fds[0].Revents
	}
	old:=send("old")
	barrier("acked")
	live:=send("live")
	if *live.SeqNum != *old.SeqNum+3 { t.Fatalf("sequence delta = %d, want 3", *live.SeqNum-*old.SeqNum) }
	sendICMPError(t,&conn,old)
	stalePoll:=poll()
	staleSO:=dut.GetSockOptInt(t,fd,unix.SOL_SOCKET,unix.SO_ERROR)
	sendICMPError(t,&conn,live)
	activePoll:=poll()
	activeSO:=dut.GetSockOptInt(t,fd,unix.SOL_SOCKET,unix.SO_ERROR)
	if activePoll!=unix.POLLERR || activeSO!=int32(unix.EHOSTUNREACH) { t.Fatalf("valid quote: poll=%d SO_ERROR=%d", activePoll,activeSO) }
	barrier("all good")
	_ = send("after both errors")
	t.Logf("STALE_ICMP ack_barrier=true seq_delta=3 stale_poll=%d stale_so=%d live_poll=%d live_so=%d new_data=true",stalePoll,staleSO,activePoll,activeSO)
}
