
func TestSACKCreditBeforeCumulativeACK(t *testing.T) {
	for _, tc := range []struct {
		name   string
		window uint16
	}{{"open", 65535}, {"closed", 0}} {
		t.Run(tc.name, func(t *testing.T) {
			dut := testbench.NewDUT(t)
			listen, port := dut.CreateListener(t, unix.SOCK_STREAM, unix.IPPROTO_TCP, 1)
			defer dut.Close(t, listen)
			wide := uint16(65535)
			conn := dut.Net.NewTCPIPv4(t, testbench.TCP{DstPort: &port, WindowSize: &wide}, testbench.TCP{SrcPort: &port})
			defer conn.Close(t)
			opts := make([]byte, 40)
			n := header.EncodeSACKPermittedOption(opts)
			n += header.EncodeMSSOption(1000, opts[n:])
			n += header.EncodeNOP(opts[n:])
			n += header.EncodeNOP(opts[n:])
			conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagSyn), Options: opts[:n]})
			synack, err := conn.Expect(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagSyn | header.TCPFlagAck)}, time.Second)
			if err != nil {
				t.Fatal(err)
			}
			if !header.ParseSynOptions(synack.Options, true).SACKPermitted {
				t.Fatal("SACK not negotiated")
			}
			// Give Linux a real RTT before testing packet credit. Otherwise its
			// independent TLP can send new data during an idle measurement.
			time.Sleep(360 * time.Millisecond)
			conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagAck)})
			fd, _ := dut.Accept(t, listen)
			defer dut.Close(t, fd)
			before := dut.GetSockOptTCPInfo(t, fd)
			mss, cwnd := int(before.SndMss), int(before.SndCwnd)
			fmt.Printf("SETUP mode=%s mss=%d cwnd=%d rtt=%d rto=%d\n", tc.name, mss, cwnd, before.RTT, before.RTO)
			if mss < 64 || mss > 1000 || cwnd < 4 || cwnd > 20 || (cwnd+6)*mss >= int(wide) {
				t.Fatalf("unexpected mss=%d cwnd=%d", mss, cwnd)
			}
			base := *conn.RemoteSeqNum(t)
			data := make([]byte, (cwnd+6)*mss)
			startSend := time.Now()
			// Write and observe each full-sized segment separately. A bulk
			// write can appear as one superpacket in a Linux packet capture.
			for i := 0; i < cwnd; i++ {
				if got := dut.Send(t, fd, data[:mss], 0); int(got) != mss {
					t.Fatalf("initial write %d = %d", i, got)
				}
				sn := uint32(base.Add(seqnum.Size(i * mss)))
				if _, err := conn.ExpectData(t, &testbench.TCP{SeqNum: &sn}, &testbench.Payload{Bytes: data[:mss]}, time.Second); err != nil {
					t.Fatalf("initial packet %d: %v", i, err)
				}
			}
			if got := dut.Send(t, fd, data[cwnd*mss:], 0); int(got) != 6*mss {
				t.Fatalf("queue tail: %d", got)
			}
			// An unsatisfiable match returns all arrivals, including repairs of older bytes.
			collect := func(wait time.Duration) []int {
				var packets []int
				frames, hit := conn.ListenForFrame(t, testbench.Layers{nil, nil, &testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagSyn | header.TCPFlagRst)}}, wait)
				if hit {
					t.Fatal("unexpected SYN|RST")
				}
				for _, frame := range frames {
					for i, l := range frame {
						h, ok := l.(*testbench.TCP)
						if !ok || *h.SrcPort != port || len(frame) <= i+1 {
							continue
						}
						p, ok := frame[i+1].(*testbench.Payload)
						if !ok || len(p.Bytes) == 0 {
							continue
						}
						off := uint32(*h.SeqNum) - uint32(base)
						if len(p.Bytes)%mss != 0 || off%uint32(mss) != 0 || uint64(off)+uint64(len(p.Bytes)) > uint64(len(data)) {
							t.Fatalf("invalid frame offset=%d bytes=%d", off, len(p.Bytes))
						}
						for j := 0; j < len(p.Bytes)/mss; j++ {
							packets = append(packets, int(off)/mss+j)
						}
					}
				}
				return packets
			}
			// No delivery/probe/recovery can be credited to the idle or duplicate gates.
			const wait = 9 * time.Millisecond
			idle := collect(min(210*time.Millisecond, time.Duration(before.RTO)*time.Microsecond/5))
			var phases [][]int
			var states []uint8
			start := time.Now()
			for _, end := range []int{2, 2, 3} {
				b := make([]byte, 40)
				n := header.EncodeNOP(b)
				n += header.EncodeNOP(b[n:])
				n += header.EncodeSACKBlocks([]header.SACKBlock{{Start: base.Add(seqnum.Size(mss)), End: base.Add(seqnum.Size(end * mss))}}, b[n:])
				conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagAck), AckNum: testbench.Uint32(uint32(base)), WindowSize: &tc.window, Options: b[:n]})
				phases = append(phases, collect(wait))
				states = append(states, dut.GetSockOptTCPInfo(t, fd).CaState)
			}
			dt := time.Since(start)
			total := time.Since(startSend)
			conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagAck), AckNum: testbench.Uint32(uint32(base.Add(seqnum.Size(cwnd * mss)))), WindowSize: &wide})
			resume := collect(25 * time.Millisecond)
			fmt.Printf("CREDIT mode=%s first=%d mss=%d idle=%v sack=%v repeat=%v extend=%v resume=%v states=%v elapsed_us=%d initial_rto_us=%d initial_rtt_us=%d send_to_last_sack_us=%d\n", tc.name, cwnd, mss, idle, phases[0], phases[1], phases[2], resume, states, dt.Microseconds(), before.RTO, before.RTT, total.Microseconds())
			if len(idle) != 0 || len(resume) == 0 || total >= time.Duration(before.RTO)*time.Microsecond {
				t.Fatal("could not isolate sender credit or resume delivery before timeout")
			}
			for _, s := range states {
				if s != linux.TCP_CA_Open && s != linux.TCP_CA_Disorder {
					t.Errorf("entered recovery: %d", s)
				}
			}
			for _, p := range phases {
				for _, i := range p {
					if i < cwnd {
						t.Errorf("unexpected repair %d before cumulative ACK", i)
					}
				}
			}
			if len(phases[1]) != 0 || (tc.window == 0 && (len(phases[0]) != 0 || len(phases[2]) != 0)) {
				t.Errorf("repeat or closed window admitted data")
			}
		})
	}
}

func TestLimitedTransmitRecovery(t *testing.T) {
	for _, small := range []bool{true, false} {
		mode := "threshold"
		if small {
			mode = "small"
		}
		t.Run(mode, func(t *testing.T) {
			dut := testbench.NewDUT(t)
			listen, port := dut.CreateListener(t, unix.SOCK_STREAM, unix.IPPROTO_TCP, 1)
			defer dut.Close(t, listen)
			wide := uint16(65535)
			conn := dut.Net.NewTCPIPv4(t, testbench.TCP{DstPort: &port, WindowSize: &wide}, testbench.TCP{SrcPort: &port})
			defer conn.Close(t)
			opts := make([]byte, 4)
			header.EncodeMSSOption(1000, opts)
			conn.ConnectWithOptions(t, opts)
			fd, _ := dut.Accept(t, listen)
			defer dut.Close(t, fd)
			dut.SetSockOpt(t, fd, unix.IPPROTO_TCP, unix.TCP_CONGESTION, []byte("reno"))
			mss := int(dut.GetSockOptTCPInfo(t, fd).SndMss)
			data := make([]byte, mss)
			write := func(buf []byte) {
				t.Helper()
				if int(dut.Send(t, fd, buf, 0)) != len(buf) {
					t.Fatal("short write")
				}
			}
			ack := func(n seqnum.Value) {
				conn.Send(t, testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagAck), AckNum: testbench.Uint32(uint32(n))})
			}
			expect := func(n seqnum.Value, timeout time.Duration) {
				t.Helper()
				if _, err := conn.ExpectData(t, &testbench.TCP{SeqNum: testbench.Uint32(uint32(n))}, &testbench.Payload{Bytes: data}, timeout); err != nil {
					t.Fatal(err)
				}
			}
			next := *conn.RemoteSeqNum(t)
			if small {
				// A real timeout followed by progress leaves a small congestion window.
				write(data)
				expect(next, time.Second)
				for {
					expect(next, 5*time.Second)
					if dut.GetSockOptTCPInfo(t, fd).CaState == linux.TCP_CA_Loss {
						break
					}
				}
				next = next.Add(seqnum.Size(mss))
				ack(next)
			}
			// Advance past any prior recovery boundary before creating the loss.
			write(data)
			expect(next, time.Second)
			next = next.Add(seqnum.Size(mss))
			ack(next)
			time.Sleep(12 * time.Millisecond)
			startInfo := dut.GetSockOptTCPInfo(t, fd)
			n := int(startInfo.SndCwnd)
			if small && (n < 2 || n > 3) || !small && (n < 6 || n > 20) {
				t.Fatalf("unexpected %s cwnd=%d", mode, n)
			}
			rto := time.Duration(startInfo.RTO) * time.Microsecond
			base := next
			pkt := func(i int) seqnum.Value { return base.Add(seqnum.Size(i * mss)) }
			start := time.Now()
			for i := 0; i < n; i++ {
				write(data)
				expect(pkt(i), time.Second)
			}
			write(make([]byte, 2*mss))
			// The first packet is lost. Only packets actually observed past that
			// hole can elicit another duplicate ACK, including new transmissions.
			arrived, dups, high, repaired := n-1, 0, n, false
			var windows, extra []int
			var repairElapsed time.Duration
			collect := func(wait time.Duration, visit func(int)) {
				frames, _ := conn.ListenForFrame(t, testbench.Layers{nil, nil, &testbench.TCP{Flags: testbench.TCPFlags(header.TCPFlagSyn | header.TCPFlagRst)}}, wait)
				for _, frame := range frames {
					for j, l := range frame {
						h, ok := l.(*testbench.TCP)
						if !ok || *h.SrcPort != port || len(frame) <= j+1 {
							continue
						}
						p, ok := frame[j+1].(*testbench.Payload)
						if !ok || len(p.Bytes) == 0 {
							continue
						}
						off := uint32(*h.SeqNum) - uint32(base)
						if off%uint32(mss) != 0 || len(p.Bytes)%mss != 0 || uint64(off)+uint64(len(p.Bytes)) > uint64((n+2)*mss) {
							t.Fatalf("unexpected payload off=%d len=%d", off, len(p.Bytes))
						}
						for i := 0; i < len(p.Bytes)/mss; i++ {
							visit(int(off)/mss + i)
						}
					}
				}
			}
			for !repaired && time.Since(start) < 3*rto+time.Second {
				if dups < 3 && dups < arrived {
					ack(base)
					dups++
				}
				collect(10*time.Millisecond, func(i int) {
					if i == 0 {
						if !repaired {
							repairElapsed = time.Since(start)
						}
						repaired = true
						return
					}
					if i != high {
						t.Fatalf("unexpected packet %d (next new %d)", i, high)
					}
					extra = append(extra, i)
					high++
					arrived++
				})
				if dups > 0 && len(windows) < dups {
					windows = append(windows, int(dut.GetSockOptTCPInfo(t, fd).SndCwnd))
				}
			}
			if !repaired {
				t.Fatalf("no repair in %v", time.Since(start))
			}
			info := dut.GetSockOptTCPInfo(t, fd)
			atRepair := high
			// Cumulatively acknowledge the repaired flight, then drain every byte
			// still queued by the sender.
			ack(pkt(high))
			for deadline := time.Now().Add(2 * time.Second); high < n+2 && time.Now().Before(deadline); {
				collect(10*time.Millisecond, func(i int) {
					if i != high {
						t.Fatalf("drain packet %d, want %d", i, high)
					}
					high++
				})
				ack(pkt(high))
			}
			fmt.Printf("LIMIT mode=%s cwnd=%d mss=%d dupacks=%d extra=%v windows=%v state=%d ssthresh=%d repaired_us=%d rto_us=%d at_repair=%d completed=%d\n", mode, n, mss, dups, extra, windows, info.CaState, info.SndSsthresh, repairElapsed.Microseconds(), rto.Microseconds(), atRepair, high)
			if high != n+2 {
				t.Fatal("stream did not finish")
			}
		})
	}
}
