
// TestRACKFreshHead asks whether a SACK implicating old data behind SND.UNA
// can cause a second transmission of a head that an RTO just retransmitted.
func TestRACKFreshHead(t *testing.T) {
	for _, schedule := range []string{"head-lost", "fresh-new-sack", "fresh-repeat-sack"} {
		t.Run(schedule, func(t *testing.T) {
			const warm = 8
			const rtt = 100 * time.Millisecond
			clk := faketime.NewManualClock()
			states := make(chan tcp.TCPSenderState, 4)
			c := context.NewWithOpts(t, context.Options{
				MTU: mtu, EnableV4: true, Clock: clk,
				Probe: func(s *tcp.TCPEndpointState) { states <- s.Sender },
			})
			defer c.Cleanup()
			e2e.EnableCUBIC(t, c)
			e2e.SetStackSACKPermitted(t, c, true)
			e2e.CreateConnectedWithSACKAndTS(c)
			first := c.IRS.Add(1 + warm*maxPayload)
			ack := func(blocks ...header.SACKBlock) tcp.TCPSenderState {
				t.Helper()
				c.SendAckWithSACK(seqnum.Value(context.TestInitialSequenceNumber).Add(1), warm*maxPayload, blocks)
				select {
				case st := <-states:
					if st.SndUna != first {
						t.Fatalf("SND.UNA = %d, want %d", st.SndUna, first)
					}
					return st
				case <-time.After(5 * time.Second):
					t.Fatal("ACK not processed")
					return tcp.TCPSenderState{}
				}
			}
			write := func(count int) {
				t.Helper()
				data := make([]byte, count*maxPayload)
				if n, err := c.EP.Write(bytes.NewReader(data), tcpip.WriteOptions{}); err != nil || int(n) != len(data) {
					t.Fatalf("Write = %d, %v", n, err)
				}
			}
			offset := func(p *buffer.View) int {
				t.Helper()
				defer p.Release()
				ip := header.IPv4(p.AsSlice())
				h := header.TCP(ip.Payload())
				if got := len(h.Payload()); got != maxPayload {
					t.Fatalf("payload length = %d, want %d", got, maxPayload)
				}
				return int(int32(h.SequenceNumber() - uint32(first)))
			}
			drain := func() []int {
				var got []int
				for p := c.GetPacketNonBlocking(); p != nil; p = c.GetPacketNonBlocking() {
					got = append(got, offset(p))
				}
				return got
			}
			expectFlight := func(begin, count int) {
				t.Helper()
				for i := begin; i < begin+count; i++ {
					if got := offset(c.GetPacket()); got != i*maxPayload {
						t.Fatalf("wire offset = %d, want %d", got, i*maxPayload)
					}
				}
			}
			write(warm)
			expectFlight(-warm, warm)
			clk.Advance(rtt)
			ack()
			write(3) // A–C: A is the first unacknowledged packet.
			expectFlight(0, 3)
			clk.Advance(5 * time.Millisecond)
			write(4) // D–G went out later; seeing D–F can mark A–C lost.
			expectFlight(3, 4)
			b := header.SACKBlock{Start: first.Add(3*maxPayload), End: first.Add(6*maxPayload)}
			stats := c.Stack().Stats().TCP
			clk.Advance(rtt)
			if schedule == "fresh-repeat-sack" {
				ack(b)
				t.Logf("BEFORE_RTO repairs=%v", drain())
			}
			var headAt tcpip.MonotonicTime
			if schedule != "head-lost" {
				for i := 0; i < 600 && stats.Timeouts.Value() == 0; i++ {
					_ = drain()
					clk.Advance(10 * time.Millisecond)
				}
				if stats.Timeouts.Value() != 1 {
					t.Fatal("RTO did not fire exactly once")
				}
				headAt = clk.NowMonotonic()
				expectFlight(0, 1)
				if extra := drain(); len(extra) != 0 {
					t.Fatalf("extra packets alongside timeout repair: %v", extra)
				}
				clk.Advance(100 * time.Microsecond)
			}
			fastBefore := stats.FastRetransmit.Value()
			st := ack(b)
			replies := drain()
			if schedule == "head-lost" && (len(replies) == 0 || replies[0] != 0) {
				t.Fatalf("head still outstanding but no head repair: %v", replies)
			}
			t.Logf("WIRE schedule=%s timeouts=%d ack_offset=0 sack=[%d,%d) us_after_RTO=%d replies=%v cwnd=%d out=%d recovery=%v sacked=%d fast_retransmit_delta=%d",
				schedule, stats.Timeouts.Value(), 3*maxPayload, 6*maxPayload, clk.NowMonotonic().Sub(headAt).Microseconds(), replies, st.SndCwnd, st.Outstanding, st.FastRecovery.Active, st.SackedOut, stats.FastRetransmit.Value()-fastBefore)
		})
	}
}
