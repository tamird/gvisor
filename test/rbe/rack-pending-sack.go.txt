
// TestRACKOldestDeadline isolates an advancing cumulative ACK from later SACK-only updates.
func TestRACKOldestDeadline(t *testing.T) {
	for _, schedule := range []string{"ack-arrives", "ack-withheld"} {
		t.Run(schedule, func(t *testing.T) {
			const warm = 8
			const rtt = 100 * time.Millisecond
			clk := faketime.NewManualClock()
			states := make(chan tcp.TCPSenderState, 8)
			c := context.NewWithOpts(t, context.Options{MTU: mtu, EnableV4: true, Clock: clk, Probe: func(s *tcp.TCPEndpointState) { states <- s.Sender }})
			defer c.Cleanup()
			e2e.EnableCUBIC(t, c)
			e2e.SetStackSACKPermitted(t, c, true)
			// Fix the RTO to the observed 200 ms deadline; this test does
			// not attempt to reproduce the WAN estimator's prior samples.
			// TCP's test context raises the minimum to 3 s. Restore the actual
			// runtime floor before constraining the upper bound for this replay.
			floor := tcpip.TCPMinRTOOption(-1)
			if err := c.Stack().SetTransportProtocolOption(tcp.ProtocolNumber, &floor); err != nil {
				t.Fatal(err)
			}
			limit := tcpip.TCPMaxRTOOption(200 * time.Millisecond)
			if err := c.Stack().SetTransportProtocolOption(tcp.ProtocolNumber, &limit); err != nil {
				t.Fatal(err)
			}
			e2e.CreateConnectedWithSACKAndTS(c)
			first := c.IRS.Add(1 + warm*maxPayload)
			left := first
			type packet struct {
				at   int
				echo uint32
			}
			var lastSentStamp uint32
			decode := func(v *buffer.View) packet {
				defer v.Release()
				h := header.TCP(header.IPv4(v.AsSlice()).Payload())
				opts := header.ParseTCPOptions(h.Options())
				if !opts.TS || len(h.Payload()) != maxPayload {
					t.Fatalf("TCP packet has TS=%v size=%d", opts.TS, len(h.Payload()))
				}
				lastSentStamp = opts.TSVal
				return packet{int(int32(h.SequenceNumber()-uint32(first))) / maxPayload, opts.TSEcr}
			}
			drain := func() []packet {
				var q []packet
				for v := c.GetPacketNonBlocking(); v != nil; v = c.GetPacketNonBlocking() {
					q = append(q, decode(v))
				}
				return q
			}
			send := func(start, count int) {
				t.Helper()
				data := make([]byte, count*maxPayload)
				if n, err := c.EP.Write(bytes.NewReader(data), tcpip.WriteOptions{}); err != nil || int(n) != len(data) {
					t.Fatalf("Write=%d,%v", n, err)
				}
				for i := 0; i < count; i++ {
					if p := decode(c.GetPacket()); p.at != start+i {
						t.Fatalf("initial send=%v want %d", p, start+i)
					}
				}
			}
			await := func() tcp.TCPSenderState {
				t.Helper()
				select {
				case st := <-states:
					if st.SndUna != left {
						t.Fatalf("UNA=%d want %d", st.SndUna, left)
					}
					return st
				case <-time.After(3 * time.Second):
					t.Fatal("TCP did not process the queued ACK")
				}
				return tcp.TCPSenderState{}
			}
			ack := func(ts, echo uint32, blocks ...header.SACKBlock) {
				var opts [40]byte
				n := header.EncodeNOP(opts[:])
				n += header.EncodeNOP(opts[n:])
				n += header.EncodeTSOption(ts, echo, opts[n:])
				if len(blocks) > 0 {
					n += header.EncodeNOP(opts[n:])
					n += header.EncodeNOP(opts[n:])
					n += header.EncodeSACKBlocks(blocks, opts[n:])
				}
				c.SendPacket(nil, &context.Headers{SrcPort: context.TestPort, DstPort: c.Port, Flags: header.TCPFlagAck, SeqNum: seqnum.Value(context.TestInitialSequenceNumber).Add(1), AckNum: left, RcvWnd: 30000, TCPOpts: opts[:n]})
			}
			block := func(i, j int) header.SACKBlock {
				return header.SACKBlock{Start: first.Add(seqnum.Size(i * maxPayload)), End: first.Add(seqnum.Size(j * maxPayload))}
			}
			send(-warm, warm)
			clk.Advance(rtt)
			ack(10001, lastSentStamp)
			await()
			send(0, 3)
			clk.Advance(5 * time.Millisecond)
			send(3, 9)
			flightStamp := lastSentStamp
			clk.Advance(rtt)
			old := []header.SACKBlock{block(10, 12), block(7, 9), block(3, 6)}
			ack(10002, flightStamp, old...)
			prior := await()
			early := drain()
			if !prior.FastRecovery.Active || len(early) == 0 || early[0].at != 0 || prior.SackedOut != 7 {
				t.Fatalf("initial recovery=%v SACKed=%d copies=%v", prior.FastRecovery.Active, prior.SackedOut, early)
			}
			repairStamp := lastSentStamp
			clk.Advance(5 * time.Millisecond)
			send(12, 2)
			clk.Advance(time.Millisecond)
			left = first.Add(maxPayload)
			ack(10003, repairStamp, old...)
			anchor := await()
			initial := drain()
			if !anchor.FastRecovery.Active || anchor.RTO != 200*time.Millisecond {
				t.Fatalf("anchor active=%v RTO=%v", anchor.FastRecovery.Active, anchor.RTO)
			}
			start := clk.NowMonotonic()
			at := func(d time.Duration) {
				t.Helper()
				delta := d - clk.NowMonotonic().Sub(start)
				if delta < 0 {
					t.Fatal("past deadline")
				}
				clk.Advance(delta)
			}
			stats := c.Stack().Stats().TCP
			// The first ACK to advance the left edge anchored the old timer.
			// A later SACK causes a repair of that new left edge, not of its predecessor.
			at(131413 * time.Microsecond)
			old[0] = block(10, 13)
			ack(10004, flightStamp, old...)
			repaired := await()
			atRepair := drain()
			if stats.Timeouts.Value() != 0 || !repaired.FastRecovery.Active || len(atRepair) == 0 || atRepair[0].at != 1 {
				t.Fatalf("needed a successful repair at SND.UNA in RACK: %v active=%v timeouts=%d prior=%v", atRepair, repaired.FastRecovery.Active, stats.Timeouts.Value(), initial)
			}
			acceptedStamp := lastSentStamp
			at(180 * time.Millisecond)
			ack(10005, flightStamp, old...)
			await()
			gapCopies := drain()
			at(200*time.Millisecond - time.Nanosecond)
			if n := stats.Timeouts.Value(); n != 0 {
				t.Fatalf("early premature timeout: %d", n)
			}
			at(200 * time.Millisecond)
			oldCount := stats.Timeouts.Value()
			oldDeadline := drain()
			if (oldCount == 1) != (len(oldDeadline) > 0 && oldDeadline[0].at == 1) {
				t.Fatalf("old counter=%d packets=%v", oldCount, oldDeadline)
			}
			at(239419 * time.Microsecond)
			var progress []packet
			if schedule == "ack-arrives" {
				left = first.Add(2 * maxPayload)
				ack(10006, acceptedStamp, old...)
				st := await()
				progress = drain()
				if st.SndUna != left {
					t.Fatalf("cumulative ACK did not advance: %d", st.SndUna)
				}
			}
			at(331413*time.Microsecond - time.Nanosecond)
			if got := stats.Timeouts.Value(); got != oldCount {
				t.Fatalf("before protected deadline: %d -> %d", oldCount, got)
			}
			at(331413 * time.Microsecond)
			fullDeadline := drain()
			fullCount := stats.Timeouts.Value()
			if fullCount > oldCount && (len(fullDeadline) == 0 || fullDeadline[0].at != 1) {
				t.Fatalf("full deadline did not repair the oldest: %v", fullDeadline)
			}
			t.Logf("DEADLINE %s rto_us=%d repaired=%v before=%v old=%d:%v progress=%v full=%d:%v", schedule, anchor.RTO.Microseconds(), atRepair, gapCopies, oldCount, oldDeadline, progress, fullCount, fullDeadline)
		})
	}
}
