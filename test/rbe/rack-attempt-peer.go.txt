// TestRACKFirstAccepted observes what actually reaches the peer when the first
// data write was refused locally. Blocks name only bytes previously observed
// on this connection.
func TestRACKFirstAccepted(t *testing.T) {
	for _, timing := range []string{"early-sack", "after-timeout"} {
		t.Run(timing, func(t *testing.T) {
			const rtt = 100*time.Millisecond
			clock:=faketime.NewManualClock()
			var reject atomic.Bool
			states:=make(chan tcp.TCPSenderState,32)
			c:=context.NewWithOpts(t,context.Options{MTU:mtu,EnableV4:true,Clock:clock,
				LinkWriteError:func(ps stack.PacketBufferList) tcpip.Error {
					for _,p:=range ps.AsSlice(){
						if p.Data().Size()!=0 && reject.Swap(false) { return &tcpip.ErrNoBufferSpace{} }
					}
					return nil
				},Probe:func(s *tcp.TCPEndpointState){ states<-s.Sender },
			})
			defer c.Cleanup()
			e2e.EnableCUBIC(t,c)
			e2e.SetStackSACKPermitted(t,c,true)
			e2e.CreateConnectedWithSACKAndTS(c)
			const warm=8
			origin:=c.IRS.Add(1+warm*maxPayload)
			write:=func(n int){ t.Helper(); b:=make([]byte,n*maxPayload); if k,e:=c.EP.Write(bytes.NewReader(b),tcpip.WriteOptions{});k!=int64(len(b))||e!=nil{t.Fatalf("write: %d,%v",k,e)} }
			drain:=func() []int { t.Helper();var ids []int
				for p:=c.GetPacketNonBlocking();p!=nil;p=c.GetPacketNonBlocking(){h:=header.TCP(header.IPv4(p.AsSlice()).Payload());if len(h.Payload())>0{ids=append(ids,1+int(int32(h.SequenceNumber()-uint32(origin)))/maxPayload)};p.Release()};return ids }
			ack:=func(n int,bs ...header.SACKBlock) tcp.TCPSenderState{t.Helper();c.SendAckWithSACK(seqnum.Value(context.TestInitialSequenceNumber).Add(1),(warm+n)*maxPayload,bs);select {case s:=<-states:return s;case <-time.After(4*time.Second):t.Fatal("no ACK observation");return tcp.TCPSenderState{}} }
			block:=func(first,last int) header.SACKBlock{return header.SACKBlock{Start:origin.Add(seqnum.Size((first-1)*maxPayload)),End:origin.Add(seqnum.Size(last*maxPayload))}}
			write(warm); if len(drain())!=warm{t.Fatal("warm sequence incomplete")};clock.Advance(rtt);window:=ack(0).SndCwnd
			reject.Store(true);write(1)
			if got:=drain();len(got)!=0||reject.Load(){t.Fatalf("first refused wire = %v",got)}
			for i:=2;i<=window;i++{write(1)}
			got:=drain();if len(got)!=window-1||got[0]!=2||got[len(got)-1]!=window{t.Fatalf("first higher flight: window=%d wires=%v",window,got)}
			for i:=0;i<30;i++{write(1)}
			got=drain()
			t.Logf("FIRST higher: window=%d extras=%v",window,got)
			seen:=map[int]bool{}
			capture:=func(ids []int){for _,n:=range ids{seen[n]=true}}
			if timing=="early-sack"{clock.Advance(rtt);st:=ack(0,block(2,window));ids:=drain();capture(ids);t.Logf("FIRST before-retry SACK 2..%d cwnd=%d out=%d packets=%v",window,st.SndCwnd,st.Outstanding,ids)}
			for i:=0;i<2400&&!seen[1];i++{clock.Advance(5*time.Millisecond);capture(drain())}
			if !seen[1]{t.Fatalf("FIRST no first on-wire copy after refusal: timeouts=%d seen=%v",c.Stack().Stats().TCP.Timeouts.Value(),seen)}
			clock.Advance(rtt/2)
			b:=block(2,window)
			st:=ack(0,b);ids:=drain();capture(ids)
			t.Logf("FIRST after first accepted SACK 2..%d cwnd=%d out=%d packets=%v timeouts=%d",window,st.SndCwnd,st.Outstanding,ids,c.Stack().Stats().TCP.Timeouts.Value())
			// If the sender emitted new high data after the first on-wire copy,
			// now (and only now) report it. Repeat otherwise-identical SACKs too.
			hi:=window;for seen[hi+1]{hi++};clock.Advance(rtt)
			st=ack(0,block(2,hi));ids=drain()
			t.Logf("FIRST followup SACK 2..%d cwnd=%d out=%d packets=%v",hi,st.SndCwnd,st.Outstanding,ids)
			// All originally observed packets can now be delivered.
			ack(window);drain()
		})
	}
}
